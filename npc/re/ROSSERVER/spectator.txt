function	script	F_CheckSpectator	{
    warp getarg(0), getarg(1), getarg(2);
    setoption Option_Invisible | Option_Xmas, 1;
    sc_start SC_NOCHAT,420000,0;
    sc_start SC_WALKSPEED,420000,2000;
    bonus bSpeedRate,50;
    sc_start SC_SPEEDUP1,420000,0;
    return;
}

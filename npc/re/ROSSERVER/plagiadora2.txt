//===== Suporta Plagiarism e Reproduce =====
//===== Data: 07 Dezembro 2025 =====
izlude,127,148,3	script	Plagiadora	10010,{
	set .@plagiarism_lv, getskilllv("RG_PLAGIARISM");
	set .@reproduce_lv, getskilllv("SC_REPRODUCE");
	
	cutin "lydia_a",2;
	mes "[ Lidia ]";
	mes "Especialista em copiar habilidades!";
	next;
	
	if (!.@plagiarism_lv && !.@reproduce_lv) {
		mes "[ Lidia ]";
		mes "Você precisa ter Plagiarism ou Reproduce!";
		cutin "", 255;
		close;
	}
	
	while(1) {
		mes "[ Lidia ]";
		mes "Status atual:";
		mes "..................................................................";
		
		// Mostra Plagiarism
		if (.@plagiarism_lv) {
			mes "[Plagiarism] Slot:";
			if (CLONE_SKILL_PLAGIARISM && CLONE_SKILL_PLAGIARISM_LV)
            {
				set .@plag_name$, getskillname(CLONE_SKILL_PLAGIARISM);
				mes "  " + .@plag_name$ + " Lv" + CLONE_SKILL_PLAGIARISM_LV;
			}
            else 
            {
				mes "  Vazio";
			}
		}
		
		// Mostra Reproduce
		if (.@reproduce_lv) {
			mes "[Reproduce] Slot:";
			if (CLONE_SKILL_REPRODUCE && CLONE_SKILL_REPRODUCE_LV) {
				set .@repro_name$, getskillname(CLONE_SKILL_REPRODUCE);
				mes "  " + .@repro_name$ + " Lv" + CLONE_SKILL_REPRODUCE_LV;
			} else {
				mes "  Vazio";
			}
		}
		mes "..................................................................";
		next;
		
		// Constrói menu
		set .@menu$, "";
		set .@opt, 1;
		
		if (.@plagiarism_lv) {
			if (CLONE_SKILL_PLAGIARISM) {
				set .@menu$, .@menu$ + "Desaprender Plagiarism:";
				set .@opt_unplag, .@opt;
			} else {
				set .@menu$, .@menu$ + "Aprender Plagiarism:";
				set .@opt_plag, .@opt;
			}
			set .@opt, .@opt + 1;
		}
		
		if (.@reproduce_lv) {
			if (CLONE_SKILL_REPRODUCE) {
				set .@menu$, .@menu$ + "Desaprender Reproduce:";
				set .@opt_unrepro, .@opt;
			} else {
				set .@menu$, .@menu$ + "Aprender Reproduce:";
				set .@opt_repro, .@opt;
			}
			set .@opt, .@opt + 1;
		}
		
		set .@menu$, .@menu$ + "Sair";
		set .@choice, select(.@menu$);
		
		// Desaprender Plagiarism
		if (.@choice == .@opt_unplag) {
			skill CLONE_SKILL_PLAGIARISM, 0, 0;
			set CLONE_SKILL_PLAGIARISM, 0;
			set CLONE_SKILL_PLAGIARISM_LV, 0;
			mes "[ Lidia ]";
			mes "Skill de Plagiarism removida!";
			next;
			continue;
		}
		
		// Aprender Plagiarism
		if (.@choice == .@opt_plag) {
			mes "[ Lidia ]";
			mes "Escolha uma skill para Plagiarism:";
			next;
			set .@idx, select(.menu$) - 1;
			
			set .@skill_id, .skill_id[.@idx];
			set .@max_lv, .skill_max_level[.@idx];
			set .@cost, .skill_cost[.@idx];
			
			set .@learn_lv, .@plagiarism_lv;
			if (.@learn_lv > .@max_lv)
				set .@learn_lv, .@max_lv;
			
			mes "[ Lidia ]";
			mes "Skill: " + .skill_name$[.@idx];
			mes "Level: " + .@learn_lv;
			mes "Slot: [Plagiarism]";
			mes "Custo: " + .@cost + " Zeny";
			next;
			
			if (select("Confirmar:Cancelar") == 2)
				continue;
			
			if (Zeny < .@cost) {
				mes "[ Lidia ]";
				mes "Zeny insuficiente!";
				next;
				continue;
			}
			
			set Zeny, Zeny - .@cost;
			set CLONE_SKILL_PLAGIARISM, .@skill_id;
			set CLONE_SKILL_PLAGIARISM_LV, .@learn_lv;
			skill CLONE_SKILL_PLAGIARISM, CLONE_SKILL_PLAGIARISM_LV, 0;
			
			mes "[ Lidia ]";
			mes "Skill aprendida com sucesso!";
			next;
			continue;
		}
		
		// Desaprender Reproduce
		if (.@choice == .@opt_unrepro) {
			skill CLONE_SKILL_REPRODUCE, 0, 0;
			set CLONE_SKILL_REPRODUCE, 0;
			set CLONE_SKILL_REPRODUCE_LV, 0;
			mes "[ Lidia ]";
			mes "Skill de Reproduce removida!";
			next;
			continue;
		}
		
		// Aprender Reproduce
		if (.@choice == .@opt_repro) {
			mes "[ Lidia ]";
			mes "Escolha uma skill para Reproduce:";
			next;
			set .@idx, select(.menu$) - 1;
			
			set .@skill_id, .skill_id[.@idx];
			set .@max_lv, .skill_max_level[.@idx];
			set .@cost, .skill_cost[.@idx];
			
			set .@learn_lv, .@reproduce_lv;
			if (.@learn_lv > .@max_lv)
				set .@learn_lv, .@max_lv;
			
			mes "[ Lidia ]";
			mes "Skill: " + .skill_name$[.@idx];
			mes "Level: " + .@learn_lv;
			mes "Slot: [Reproduce]";
			mes "Custo: " + .@cost + " Zeny";
			next;
			
			if (select("Confirmar:Cancelar") == 2)
				continue;
			
			if (Zeny < .@cost) {
				mes "[ Lidia ]";
				mes "Zeny insuficiente!";
				next;
				continue;
			}
			
			set Zeny, Zeny - .@cost;
			set CLONE_SKILL_REPRODUCE, .@skill_id;
			set CLONE_SKILL_REPRODUCE_LV, .@learn_lv;
			skill CLONE_SKILL_REPRODUCE, CLONE_SKILL_REPRODUCE_LV, 0;
			
			mes "[ Lidia ]";
			mes "Skill aprendida com sucesso!";
			next;
			continue;
		}
		
		break;
	}
	
	cutin "", 255;
	close;

OnInit:
	setarray .skill_name$, "Avanco Ofensivo", "Laminas Retalhadoras", "Nevasca", "Erva Daninha", "Planta Infernal", "Revelacao";
	setarray .skill_id, 1001, 2022, 89, 2483, 2490, 24;
	setarray .skill_max_level, 1, 5, 10, 10, 5, 1;
	setarray .skill_cost, 1, 1, 1, 1, 1, 1;
	
	set .menu$, "";
	set .skill_size, getarraysize(.skill_name$);
	set .@i, 0;
	while (.@i < .skill_size) {
		set .menu$, .menu$ + "[ Lv " + .skill_max_level[.@i] + " ] " + .skill_name$[.@i] + ":";
		set .@i, .@i + 1;
	}
	end;
}

OnPCLoadMapEvent:

		skill CLONE_SKILL_PLAGIARISM, 0, 0;
		skill CLONE_SKILL_PLAGIARISM, CLONE_SKILL_PLAGIARISM_LV, 0;
	}
	
	if (getskilllv("SC_REPRODUCE") && CLONE_SKILL_REPRODUCE && CLONE_SKILL_REPRODUCE_LV) {
		skill CLONE_SKILL_REPRODUCE, 0, 0;
		skill CLONE_SKILL_REPRODUCE, CLONE_SKILL_REPRODUCE_LV, 0;
	}
	end;
}

// Battleground Engine
// *********************************************************************

-	script	KvM80_BG	-1,{
	end;

OnGuillaumeJoin:
OnCroixJoin:
	donpcevent "KvM80_BG::OnReadyCheck";
	end;

OnGuillaumeQuit:
	set BG_Delay_Tick, gettimetick(2) + 2000;
OnGuillaumeDie:
	if( $@KvM80BG == 2 )
	{
		set .Guillaume_Count, .Guillaume_Count - 1;
		set .Croix_Score, .Croix_Score + 1;
		bg_updatescore "ros2024",.Guillaume_Score,.Croix_Score;
		if( (.Guillaume_Count) < 1 ){
			donpcevent "KvM80_BG::OnCroixWin";
		}
	}
	end;

OnCroixQuit:
	set BG_Delay_Tick, gettimetick(2) + 2000;
OnCroixDie:
	if( $@KvM80BG == 2 )
	{
		set .Croix_Count, .Croix_Count - 1;
		set .Guillaume_Score, .Guillaume_Score + 1;
		bg_updatescore "ros2024",.Guillaume_Score,.Croix_Score;
		if( (.Croix_Count) < 1 ){
			donpcevent "KvM80_BG::OnGuillaumeWin";
		}
	}
	end;

OnReadyCheck:
	if( $@KvM80BG )
		end;
	set .@Guillaume, getwaitingroomstate(0,"KvM80R_Guillaume");
	set .@Croix, getwaitingroomstate(0,"KvM80R_Croix");

	if((.@Guillaume < $@jogadoresblue) || (.@Croix < $@jogadoresred) )
	{
		if( .@Guillaume > 3 && .@Croix > 3 && !agitcheck() && $@KvM_Flood < gettimetick(2) )
		{
			announce "Arena 1 - ros2024 | Blue Team: " + .@Guillaume + "/"+$@jogadoresblue+", Red Team: " + .@Croix + "/"+$@jogadoresred+"",0,0xFFA500;
			set $@KvM_Flood, gettimetick(2) + 15;
		}
		else
			mapannounce "quiz_02","Arena 1 - ros2024 | Blue Team: " + .@Guillaume + "/"+$@jogadoresblue+", Red Team: " + .@Croix + "/"+$@jogadoresred+"",1,0xFFA500;
		end;
	}
		
	if ((.@Guillaume == 0) || (.@Croix == 0)) {
		end;
	}
	
	set $@KvM80BG, 1; // Starting
	donpcevent "KvM80R_Croix::OnEnterBG";
	donpcevent "KvM80R_Guillaume::OnEnterBG";
	donpcevent "KvM80_BG::OnStart";
	end;

OnStart:
	announce "- Duas equipes entraram na [ Arena 1 - ros2024 ]. -", bc_yellow;
	set $@KvM80BG_Victory, 0;
	// Warp Teams
	bg_warp $@KvM80BG_id1,"ros2024",64,92;
	bg_warp $@KvM80BG_id2,"ros2024",195,92;

	if ($@guildsInArena > 0) {
        $@guildsInArena--;
    }
	if ($@guildsInArena > 0) {
        $@guildsInArena--;
    }

	setwall "ros2024",179,95,10,4,0,"a";
	setwall "ros2024",80,95,10,4,0,"b";
	// setwall "ros2024",229,113,8,4,0,"batc01wall_ae01";
	// setwall "ros2024",113,113,8,4,0,"batc01wall_ae02";

	sleep 100;
	// ScoreBoard
	set .Guillaume_Score, 0;
	set .Croix_Score, 0;
	initnpctimer;
	bg_updatescore "ros2024",.Guillaume_Score,.Croix_Score;
	end;

OnTimer5000:
	mapannounce "ros2024","- The match will begin in 25 seconds! -", bc_yellow;
	end;

OnTimer25000:
	mapannounce "ros2024","- The match will begin in 5 seconds! -", bc_yellow;
	end;

OnTimer26000:
	mapannounce "ros2024","- The match will begin in 4 seconds! -", bc_yellow;
	end;

OnTimer27000:
	mapannounce "ros2024","- The match will begin in 3 seconds! -", bc_yellow;
	end;

OnTimer28000:
	mapannounce "ros2024","- The match will begin in 2 seconds! -", bc_yellow;
	end;
	
OnTimer29000:
	mapannounce "ros2024","- The match will begin in 1 second! -", bc_yellow;
	end;

OnTimer30000:
	// Team Members
	set .Guillaume_Count, bg_get_data($@KvM80BG_id1, 0);
	set .Croix_Count, bg_get_data($@KvM80BG_id2, 0);
	if( (.Guillaume_Count < $@jogadoresblue) || (.Croix_Count < $@jogadoresred) )
	{
		set $@KvM80BG_Victory, 3;
		set $@KvM80BG, 3;
		mapannounce "ros2024","- There are not enough players to start the match. -", bc_yellow;
		stopnpctimer;
		sleep 2000;
		donpcevent "KvM80_BG::OnStop";
		end;
	}

	set $@KvM80BG, 2; // Playing
	delwall "a";
	delwall "b";
	// delwall "batc01wall_ae01";
	// delwall "batc01wall_ae02";
	end;

OnTimer30001:
	mapannounce "ros2024","- The match started! Good luck to all. -", bc_yellow;
killmonsterall "ros2024";
	end;

OnTimer420000:
//OnTimer60000:
	mapannounce "ros2024","- The match will end in 30 seconds! -", bc_yellow;
	end;

OnTimer450000:
//OnTimer90000:
	if( .Croix_Count > .Guillaume_Count )
		donpcevent "KvM80_BG::OnCroixWin";
	else if( .Croix_Count < .Guillaume_Count )
		donpcevent "KvM80_BG::OnGuillaumeWin";
	else
	{ // Draw Game
		set $@KvM80BG, 3;
		set $@KvM80BG_Victory, 3;

		stopnpctimer;
		sleep 2000;
		announce "- As duas equipes empataram! -", bc_yellow;
		donpcevent "KvM80_BG::OnStop";
	}
	end;

OnGuillaumeWin:
	set $@KvM80BG, 3;
	set $@KvM80BG_Victory, 1;

	stopnpctimer;
	sleep 1000;	
	announce "- Blue Team VENCEU! -",bc_all,0x00bfff;
	sleep 3000;
	donpcevent "KvM80_BG::OnStop";
	end;

OnCroixWin:
	set $@KvM80BG, 3;
	set $@KvM80BG_Victory, 2;

	stopnpctimer;
	sleep 1000;
	announce "- Red Team VENCEU! -",bc_all,0xff6347;
	sleep 3000;
	donpcevent "KvM80_BG::OnStop";
	end;

OnStop:
	if ($spectator_count9 > 0) {
       set $spectator_count9, $spectator_count9 - 1;
    }
	if ($spectator_count9 > 0) {
       set $spectator_count9, $spectator_count9 - 1;
    }
	// Warp Teams
	mapwarp "ros2024","izlude",128,135;
	donpcevent "KvM80_BG_Out::OnBegin";
	end;

OnReset:
	stopnpctimer;
	stopnpctimer "KvM80_BG_Out";
	set .Croix_Count, 0;
	set .Guillaume_Count, 0;
	set .Croix_Score, 0;
	set .Guillaume_Score, 0;
	set $@KvM80BG_Victory, 0;
	if( $@KvM80BG_id1 ) { bg_destroy $@KvM80BG_id1; set $@KvM80BG_id1, 0; }
	if( $@KvM80BG_id2 ) { bg_destroy $@KvM80BG_id2; set $@KvM80BG_id2, 0; }
	sleep 2000;
	mapwarp "ros2024","izlude",128,135;
	sleep 2000;
	maprespawnguildid "ros2024",0,3; // Just in case someone else
	bg_updatescore "ros2024",0,0;
	sleep 2000;
	set $@KvM80BG, 0;
	donpcevent "KvM80_BG::OnReadyCheck"; // Maybe a game is ready to start
	end;
}

-	script	KvM80_BG_Out	-1,{
	end;

OnBegin:
	initnpctimer;
	end;

OnTimer50000:
	mapannounce "ros2024","- The match will end in 10 seconds! -",1,0x696969;
	end;
OnTimer5000:
	donpcevent "KvM80_BG::OnReset";
	end;
}

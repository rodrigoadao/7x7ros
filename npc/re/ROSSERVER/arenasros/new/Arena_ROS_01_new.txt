OnGuillaumeQuit:
	set BG_Delay_Tick, gettimetick(2) + 2000;
	
	// Se estiver em preparacao (estado 1), cancelar a partida
	if ($@KvM99BG == 1) {
		set $@KvM99BG_Victory, 3;
		set $@KvM99BG, 3;
		mapannounce "ros2024","- Partida cancelada: um jogador do Time Azul saiu durante a prepara��o! -", bc_yellow;
		donpcevent "Arena7x7_Ranking_Hook_new::OnMatchCancel";
		donpcevent "KvM99_BG::OnReset";  // Usar OnReset para destruir BG imediatamente
		end;
	}
	
	// Se a partida esta ativa (estado 2), aplicar penalidade de desertor
	if ($@KvM99BG == 2) {
		// Decrementar contador de jogadores vivos do time azul
		if (.Guillaume_Count > 0) {
			set .Guillaume_Count, .Guillaume_Count - 1;
		}
		// Atualizar placar
		bg_updatescore "ros2024",.Guillaume_Count,.Croix_Count;
		// Aplicar penalidade de desertor (-3 pontos)
		if ($@arena1_match_id > 0) {
			arena7x7_leave(1); // 1 = desertor com penalidade
		}
		// Verificar se o time azul foi eliminado
		if (.Guillaume_Count < 1) {
			donpcevent "KvM99_BG::OnCroixWin";
		}
	}
	end;

			set .Guillaume_Count, .Guillaume_Count - 1;
		}
		// Incrementar pontua??o do time vermelho (1 ponto por kill)

OnStart:
	announce "- Duas equipes entraram na [ Arena 1 - ros2024 ]. -", bc_yellow;
	set $@KvM99BG_Victory, 0;
	
	// Criar partida no sistema de ranking ANTES do warp (para rastrear desertores durante preparo)
	donpcevent "Arena7x7_Ranking_Hook_new::OnMatchCreate";
	
	// Remover buffs de todos em quiz_02 antes de teleportar
	getmapunits BL_PC, "quiz_02", .@units;
	for (set .@i, 0; .@i < getarraysize(.@units); set .@i, .@i + 1) {
		if (isloggedin(.@units[.@i])) {
			attachrid .@units[.@i];
			sc_end SC_ALL;
			setoption 0x40 | Option_Invisible | Option_Xmas, 0;
			sc_end SC_FIGHTINGSPIRIT;
			sc_end SC_LUXANIMA;
			sc_end SC_GIANTGROWTH;
			sc_end SC_STONEHARDSKIN;
			sc_end SC_VITALITYACTIVATION;
			sc_end SC_ABUNDANCE;
			sc_end SC_CRUSHSTRIKE;
			sc_end SC_REFRESH;
			sc_end SC_MILLENNIUMSHIELD;
			delspiritball 15;
			deleteelemental;
			sc_end SC_ALL;
			sc_end SC_SAVAGE_STEAK;
			sc_end SC_COCKTAIL_WARG_BLOOD;
			sc_end SC_MINOR_BBQ;
			sc_end SC_SIROMA_ICE_TEA;
			sc_end SC_DROCERA_HERB_STEAMED;
			sc_end SC_PUTTI_TAILS_NOODLES;
			sc_end SC_STOMACHACHE;
			sc_end SC_VITATA_500;
			sc_end SC_EXTRACT_SALAMINE_JUICE;
		}
	}

	donpcevent "Arena_AutoRestock::OnRestock";
	
	// Warp Teams
	bg_warp $@KvM99BG_id1,"ros2024",64,92;
	bg_warp $@KvM99BG_id2,"ros2024",195,92;

	if ($@guildsInArena > 0) {
        $@guildsInArena--;
    }
	if ($@guildsInArena > 0) {
        $@guildsInArena--;
    }

	setwall "ros2024",179,95,10,4,0,"a";
	setwall "ros2024",80,95,10,4,0,"b";

	sleep 100;
	// ScoreBoard
	set .Guillaume_Score, 0;
	set .Croix_Score, 0;
	
	// Obter quantidade de jogadores em cada time IMEDIATAMENTE ap?s warp
	set .Guillaume_Count, bg_get_data($@KvM99BG_id1, 0);
	set .Croix_Count, bg_get_data($@KvM99BG_id2, 0);
	
	// Usar Score (azul x vermelho) e Count (vivos) no placar
	// O placar mostra: Score Azul | Vivos Azul vs Vivos Vermelho | Score Vermelho
	// bg_updatescore usa os valores como placar esquerdo e direito
	// Mas precisamos mostrar jogadores vivos, ent?o usaremos Count
	bg_updatescore "ros2024",.Guillaume_Count,.Croix_Count;
	
	initnpctimer;
	end;

OnTimer5000:
	mapannounce "ros2024","- The match will begin in 25 seconds! -", bc_yellow;

OnTimer25000:
	// VERIFICA??O antecipada de jogadores (5 segundos antes)
	set .@blue_check, bg_get_data($@KvM99BG_id1, 0);
	set .@red_check, bg_get_data($@KvM99BG_id2, 0);
	
	if( (.@blue_check < $@jogadoresblue) || (.@red_check < $@jogadoresred) )
	{
		set $@KvM99BG_Victory, 3;
		set $@KvM99BG, 3;
		
		set .@missing_blue, $@jogadoresblue - .@blue_check;
		set .@missing_red, $@jogadoresred - .@red_check;
		
		if (.@missing_blue > 0 && .@missing_red > 0) {
			mapannounce "ros2024","- Partida cancelada: jogadores desconectaram de ambos os times! -", bc_yellow;
		} else if (.@missing_blue > 0) {
			mapannounce "ros2024","- Partida cancelada: " + .@missing_blue + " jogador(es) do Time Azul desconectou! -", bc_yellow;
		} else {
			mapannounce "ros2024","- Partida cancelada: " + .@missing_red + " jogador(es) do Time Vermelho desconectou! -", bc_yellow;
		}
		
		donpcevent "Arena7x7_Ranking_Hook_new::OnMatchCancel";
		stopnpctimer;
		sleep 2000;
		donpcevent "KvM99_BG::OnReset";  // Usar OnReset para destruir BG imediatamente
		end;
	}
	
	mapannounce "ros2024","- The match will begin in 5 seconds! -", bc_yellow;
	end;

OnTimer30000:
	// ============================================================================
	// VERIFICA??O CR?TICA: Obter contagem REAL de players nos battlegrounds
	// Usa bg_get_data() para garantir precis?o no momento exato do in?cio
	// ============================================================================
	set .@blue_count_real, bg_get_data($@KvM99BG_id1, 0);
	set .@red_count_real, bg_get_data($@KvM99BG_id2, 0);
	
	// Verifica??o de jogadores suficientes para iniciar
	if( (.@blue_count_real < $@jogadoresblue) || (.@red_count_real < $@jogadoresred) )
	{
		set $@KvM99BG_Victory, 3;
		set $@KvM99BG, 3;
		
		// Obter nomes das guilds para mensagens
		if ($@BlueGuildName$ == "") set .@bluename$, "Blue Team"; else set .@bluename$, $@BlueGuildName$;
		if ($@RedGuildName$ == "") set .@redname$, "Red Team"; else set .@redname$, $@RedGuildName$;
		
		// Detectar qual time perdeu jogadores
		set .@missing_blue, $@jogadoresblue - .@blue_count_real;
		set .@missing_red, $@jogadoresred - .@red_count_real;
		
		if (.@missing_blue > 0 && .@missing_red > 0) {
			mapannounce "ros2024","- Partida cancelada: jogadores desconectaram de ambos os times! -", bc_yellow;
			mapannounce "ros2024","- " + .@bluename$ + ": " + .@blue_count_real + "/" + $@jogadoresblue + " | " + .@redname$ + ": " + .@red_count_real + "/" + $@jogadoresred + " -", bc_yellow;
		} else if (.@missing_blue > 0) {
			mapannounce "ros2024","- Partida cancelada: " + .@missing_blue + " jogador(es) da Guild " + .@bluename$ + " desconectou! -", bc_yellow;
			mapannounce "ros2024","- " + .@bluename$ + ": " + .@blue_count_real + "/" + $@jogadoresblue + " players presentes -", bc_yellow;
		} else {
			mapannounce "ros2024","- Partida cancelada: " + .@missing_red + " jogador(es) da Guild " + .@redname$ + " desconectou! -", bc_yellow;
			mapannounce "ros2024","- " + .@redname$ + ": " + .@red_count_real + "/" + $@jogadoresred + " players presentes -", bc_yellow;
		}
		
		// Cancelar a partida no sistema de ranking (se foi criada)
		donpcevent "Arena7x7_Ranking_Hook_new::OnMatchCancel";
		
		stopnpctimer;
		sleep 2000;
		donpcevent "KvM99_BG::OnStop";
		end;
	}
	
	// Atualizar contadores com valores reais antes de iniciar
	set .Guillaume_Count, .@blue_count_real;
	set .Croix_Count, .@red_count_real;
	
	set $@KvM99BG, 2; // Playing - Partida iniciada
	
	// Iniciar a partida no sistema de ranking (jogadores j? foram adicionados em OnMatchCreate)
	donpcevent "Arena7x7_Ranking_Hook_new::OnMatchStart";
	
	delwall "a";
	delwall "b";
	// delwall "batc01wall_ae01";
	// delwall "batc01wall_ae02";
	end;

OnTimer450000:

		donpcevent "KvM99_BG::OnCroixWin";
	else if( .Croix_Count < .Guillaume_Count )
		donpcevent "KvM99_BG::OnGuillaumeWin";
	else
	{ // Draw Game
		set $@KvM99BG_Victory, 3;
		donpcevent "Arena7x7_Ranking_Hook_new::OnDraw";
		
		stopnpctimer;
		sleep 2000;
		announce "- As duas equipes empataram! -", bc_yellow;
		donpcevent "KvM99_BG::OnStop";
	}
	end;

	 donpcevent "Arena7x7_Ranking_Hook_new::OnRedWin"; 
	
	stopnpctimer;
	sleep 1000;
	sleep 3000;
	donpcevent "KvM99_BG::OnStop";
	end;

OnStop:
	if ($spectator_count9 > 0) {
       set $spectator_count9, $spectator_count9 - 1;
    }
	if ($spectator_count9 > 0) {
       set $spectator_count9, $spectator_count9 - 1;
    }
	// Restaurar todos os tumbas para jogadores normais
	if ($@arena1_match_id > 0) {
		arena7x7_restoreall($@arena1_match_id);
	}
	//
// Restock automatico ANTES do warp (jogadores ainda em ros2024)
	donpcevent "Arena_AutoRestock::OnRestock";
	atcommand "@repairall";
	//
	sleep 500;
	// Teleportar membros do BG (vivos)
	if ($@KvM99BG_id1) bg_warp $@KvM99BG_id1,"izlude",128,135;
	if ($@KvM99BG_id2) bg_warp $@KvM99BG_id2,"izlude",128,135;
	sleep 500;
	// Teleportar ex-tumbas que foram restauradas (nao estao mais no BG)
	mapwarp "ros2024","izlude",128,135;
	donpcevent "KvM99_BG_Out::OnBegin";
	end;

OnReset:
	donpcevent "Arena7x7_Ranking_Hook_new::OnMatchCancel";
	stopnpctimer;
	stopnpctimer "KvM99_BG_Out";
	set .Croix_Count, 0;
	set .Guillaume_Count, 0;
	set .Croix_Score, 0;
	set .Guillaume_Score, 0;
	set $@KvM99BG_Victory, 0;
	// Se a partida foi cancelada (state==3), teleportar jogadores antes de destruir BG
	if( $@KvM99BG == 3 ) {
		bg_warp $@KvM99BG_id1,"izlude",128,135;
		bg_warp $@KvM99BG_id2,"izlude",128,135;
		mapwarp "ros2024","izlude",128,135;
	}
	// Destruir BG
	if( $@KvM99BG_id1 ) { bg_destroy $@KvM99BG_id1; set $@KvM99BG_id1, 0; }
	if( $@KvM99BG_id2 ) { bg_destroy $@KvM99BG_id2; set $@KvM99BG_id2, 0; }
	sleep 2000;
	maprespawnguildid "ros2024",0,3; // Just in case someone else
	bg_updatescore "ros2024",0,0;
	sleep 2000;
	set $@KvM99BG, 0;
	donpcevent "KvM99_BG::OnReadyCheck"; // Maybe a game is ready to start
	end;
}

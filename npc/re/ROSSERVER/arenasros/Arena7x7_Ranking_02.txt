	end;

// Chamado quando uma partida ? criada (durante prepara??o - 30s)
// S? cria a partida sem adicionar jogadores ainda
OnMatchCreate:
	// Criar partida no sistema de ranking
	set $@arena02_match_id, arena7x7_create("ros2025_1");
	
	if ($@arena02_match_id == 0) {
		//debugmes "[Arena7x7] ERRO: Falha ao criar partida no sistema de ranking";
		end;
	}
	
	//debugmes "[Arena7x7] Partida " + $@arena02_match_id + " criada";
	end;

// Chamado quando a partida REALMENTE inicia (ap?s 30s de prepara??o)
// Adiciona jogadores ? partida j? criada
OnMatchStart:
	if ($@arena02_match_id == 0) {
		//debugmes "[Arena7x7] ERRO: Tentativa de iniciar partida sem match_id";
		end;
	}
	
	// Pegar informacoes das guilds dos times
	set $@arena02_blue_guild$, "";
	set $@arena02_red_guild$, "";
	
	// Adicionar jogadores do Time Azul e pegar nome da guild
	set .@bg_id, $@KvM81BG_id1;
	set .@count, bg_get_data(.@bg_id, 0);
	
	for (set .@i, 0; .@i < .@count; set .@i, .@i + 1) {
		// Usar mapreg para pegar os char_ids
		bg_get_data(.@bg_id, 1);  // Popula $@arenamembers
		set .@aid, $@arenamembers[.@i];
		if (.@aid > 0) {
			if (attachrid(.@aid)) {
				// Pegar nome da guild se ainda nao temos
				if ($@arena02_blue_guild$ == "" && getcharid(2) > 0) {
					set $@arena02_blue_guild$, getguildname(getcharid(2));
				}
				
				if (arena7x7_join($@arena02_match_id, 1, (.@i == 0 ? 1 : 0))) {
					//debugmes "[Arena7x7] Jogador " + strcharinfo(0) + " adicionado ao Time Azul";
				}
				detachrid;
			}
		}
	}
	
	// Adicionar jogadores do Time Vermelho e pegar nome da guild
	set .@bg_id, $@KvM81BG_id2;
	set .@count, bg_get_data(.@bg_id, 0);
	
	for (set .@i, 0; .@i < .@count; set .@i, .@i + 1) {
		bg_get_data(.@bg_id, 1);  // Popula $@arenamembers
		set .@aid, $@arenamembers[.@i];
		if (.@aid > 0) {
			if (attachrid(.@aid)) {
				// Pegar nome da guild se ainda nao temos
				if ($@arena02_red_guild$ == "" && getcharid(2) > 0) {
					set $@arena02_red_guild$, getguildname(getcharid(2));
				}
				
				if (arena7x7_join($@arena02_match_id, 2, (.@i == 0 ? 1 : 0))) {
					//debugmes "[Arena7x7] Jogador " + strcharinfo(0) + " adicionado ao Time Vermelho";
				}
				detachrid;
			}
		}
	}
	
	//debugmes "[Arena7x7] Todos jogadores adicionados ? partida " + $@arena02_match_id;
	
	// Iniciar a partida no sistema de ranking
	arena7x7_start($@arena02_match_id);
	
	// Recuperar os nomes das guilds do sistema C++
	set .@blue_guild$, arena7x7_getguildname($@arena02_match_id, 1);
	set .@red_guild$, arena7x7_getguildname($@arena02_match_id, 2);
	
	// Se n?o temos do C++, usar as vari?veis do script
	if (.@blue_guild$ == "") set .@blue_guild$, $@arena02_blue_guild$;
	if (.@red_guild$ == "") set .@red_guild$, $@arena02_red_guild$;
	
	// Anuncio GLOBAL do inicio da partida (compacto)
	if (.@blue_guild$ != "" && .@red_guild$ != "") {
		//announce "aYt " + .@blue_guild$ + " vs " + .@red_guild$ + "Partida" + $@arena02_match_id + " iniciada na arena ros2025_1 (GH)",bc_all,0xFFD700;
		announce "Iniciada a partida " + $@arena02_match_id + " na arena ros2025_1 (GH)" + ": " + .@blue_guild$ + " vs " + .@red_guild$,bc_all,0xFFD700;
	} else {
		announce "BLUE TEAM vs RED TEAM - Partida " + $@arena02_match_id + " iniciada na arena ros2025_1 (GH)",bc_all,0xFFD700;
	}
	
	//debugmes "[Arena7x7] Partida " + $@arena02_match_id + " iniciada";
	end;

// Chamado quando Time Azul vence
OnBlueWin:
	if ($@arena02_match_id == 0) end;
	
	set .@blue_score, getvariableofnpc(.Guillaume_Score, "KvM81_BG");
	set .@red_score, getvariableofnpc(.Croix_Score, "KvM81_BG");
	
	arena7x7_finish($@arena02_match_id, 1, .@blue_score, .@red_score);
	
	// Recuperar nomes das guilds do sistema C++
	set .@blue_guild$, arena7x7_getguildname($@arena02_match_id, 1);
	set .@red_guild$, arena7x7_getguildname($@arena02_match_id, 2);
	if (.@blue_guild$ == "") set .@blue_guild$, "BLUE TEAM";
	if (.@red_guild$ == "") set .@red_guild$, "RED TEAM";
	
	// Anuncio GLOBAL do fim da partida (compacto)
	//announce "aYt " + .@blue_guild$ + " venceu (" + .@blue_score + "x" + .@red_score + ") a Partida " + $@arena02_match_id + " na arena ros2025_1 (GH)",bc_all,0x00BFFF;
	announce .@blue_guild$ + " venceu (" + .@blue_score + "x" + .@red_score + ") a Partida " + $@arena02_match_id + " na arena ros2025_1 (GH)",bc_all,0x00BFFF;

	// Enviar estatisticas para os jogadores
	callfunc("F_SendMatchStats", $@arena02_match_id, 1);
	
	//debugmes "[Arena7x7] Partida " + $@arena02_match_id + " finalizada - Vitoria Time Azul";
	set $@arena02_match_id, 0;
	set $@arena02_blue_guild$, "";
	set $@arena02_red_guild$, "";
	end;

// Chamado quando Time Vermelho vence
OnRedWin:
	if ($@arena02_match_id == 0) end;
	
	set .@blue_score, getvariableofnpc(.Guillaume_Score, "KvM81_BG");
	set .@red_score, getvariableofnpc(.Croix_Score, "KvM81_BG");
	
	arena7x7_finish($@arena02_match_id, 2, .@blue_score, .@red_score);
	
	// Recuperar nomes das guilds do sistema C++
	set .@blue_guild$, arena7x7_getguildname($@arena02_match_id, 1);
	set .@red_guild$, arena7x7_getguildname($@arena02_match_id, 2);
	if (.@blue_guild$ == "") set .@blue_guild$, "BLUE TEAM";
	if (.@red_guild$ == "") set .@red_guild$, "RED TEAM";
	
	// Anuncio GLOBAL do fim da partida (compacto)
	//announce "aYt " + .@red_guild$ + " venceu (" + .@red_score + "x" + .@blue_score + ") a Partida " + $@arena02_match_id + " na arena ros2025_1 (GH)",bc_all,0xFF6347;
	announce .@red_guild$ + " venceu (" + .@red_score + "x" + .@blue_score + ") a Partida " + $@arena02_match_id + " na arena ros2025_1 (GH)",bc_all,0xFF6347;

	// Enviar estatisticas para os jogadores
	callfunc("F_SendMatchStats", $@arena02_match_id, 2);
	
	//debugmes "[Arena7x7] Partida " + $@arena02_match_id + " finalizada - Vitoria Time Vermelho";
	set $@arena02_match_id, 0;
	set $@arena02_blue_guild$, "";
	set $@arena02_red_guild$, "";
	end;

// Chamado quando ha empate
OnDraw:
	if ($@arena02_match_id == 0) end;
	
	set .@blue_score, getvariableofnpc(.Guillaume_Score, "KvM81_BG");
	set .@red_score, getvariableofnpc(.Croix_Score, "KvM81_BG");
	
	arena7x7_finish($@arena02_match_id, 0, .@blue_score, .@red_score);
	
	// Anuncio GLOBAL do fim da partida (compacto)
	//announce "aYt Empate (" + .@blue_score + "x" + .@red_score + ") na Partida " + $@arena02_match_id + " na arena ros2025_1 (GH)",bc_all,0xFFFF00;
	announce "Empate (" + .@blue_score + "x" + .@red_score + ") na Partida " + $@arena02_match_id + " na arena ros2025_1 (GH)",bc_all,0xFFFF00;

	// Enviar estatisticas para os jogadores
	callfunc("F_SendMatchStats", $@arena02_match_id, 0);
	
	//debugmes "[Arena7x7] Partida " + $@arena02_match_id + " finalizada - Empate";
	set $@arena02_match_id, 0;
	set $@arena02_blue_guild$, "";
	set $@arena02_red_guild$, "";
	end;

// Chamado quando a partida e cancelada
OnMatchCancel:
	if ($@arena02_match_id == 0) end;
	
	arena7x7_cancel($@arena02_match_id);
	
	announce "ARENA 7x7 - Partida " + $@arena02_match_id + " CANCELADA",bc_all,0xFF0000;
	
	//debugmes "[Arena7x7] Partida " + $@arena02_match_id + " cancelada";
	set $@arena02_match_id, 0;
	set $@arena02_blue_guild$, "";
	set $@arena02_red_guild$, "";
	end;

// Chamado quando um jogador sai (desertor) - REMOVIDO arena7x7_leave sem RID
// O sistema j? cancela a partida automaticamente via OnTimer25000 e OnTimer30000
OnPlayerDeserter:
	// N?o h? RID attachado quando OnGuillaumeQuit/OnCroixQuit ? chamado
	// O cancelamento ? tratado pelas verifica??es de contagem nos timers
	end;
}

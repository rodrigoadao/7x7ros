//===== rAthena Script =======================================
//= Battleground: Fila 4v4
//===== Description: =========================================
//= Depende de bg_arenas_registro.txt (precisa estar carregado antes
//= - arrays $@bg3v3_arena_* e as mapflags das 7 arenas). Uma sala de
//= espera pra 8 pessoas; ao encher, sorteia 4 pra cada time e manda
//= pra uma arena livre entre as 7 compartilhadas.
//============================================================

izlude,143,129,5	script	Fila 4v4#bg_3v3	88,{
	end;
OnInit:
	.teamsize     = 4;
	.eventlasting = 10*60;
	setarray .rewarditem[0], 501, 15;
	sleep 1;
OnWaitOpen:
	waitingroom "Fila 4v4 (8)", .teamsize *2 +1, strnpcinfo(0)+"::OnMatchStart", .teamsize *2;
	end;
OnMatchStart:
	.@n = .teamsize *2;
	if ( getwaitingroomstate(0) < .@n )
		end;

	.@free = 0;
	while ( .@free == 0 ) {
		for ( .@i = 0; .@i < $@bg3v3_arena_count; ++.@i ) {
			if ( !$@bg3v3_arena_busy[.@i] ) {
				.@freeidx[.@free] = .@i;
				.@free = .@free +1;
			}
		}
		if ( .@free == 0 )
			sleep 3000;
	}
	.@aidx = .@freeidx[ rand(0, .@free -1) ];
	$@bg3v3_arena_busy[.@aidx] = 1;

	.arenaname$ = $@bg3v3_arena_map$[.@aidx];
	.redx  = $@bg3v3_arena_redx[.@aidx];
	.redy  = $@bg3v3_arena_redy[.@aidx];
	.bluex = $@bg3v3_arena_bluex[.@aidx];
	.bluey = $@bg3v3_arena_bluey[.@aidx];

	.red  = bg_create( .arenaname$, .redx,  .redy,  strnpcinfo(0)+"::OnRedQuit",  strnpcinfo(0)+"::OnRedDead" );
	.blue = bg_create( .arenaname$, .bluex, .bluey, strnpcinfo(0)+"::OnBlueQuit", strnpcinfo(0)+"::OnBlueDead" );

	warpwaitingpc .arenaname$, .redx, .redy, .@n;
	delwaitingroom;

	copyarray .@ids[0], $@warpwaitingpc[0], .@n;
	for ( .@i = .@n -1; .@i > 0; --.@i ) {
		.@j = rand(0, .@i);
		.@tmp = .@ids[.@i];
		.@ids[.@i] = .@ids[.@j];
		.@ids[.@j] = .@tmp;
	}
	for ( .@i = 0; .@i < .teamsize; ++.@i ) {
		attachrid .@ids[.@i];
		bg_join .red, .arenaname$, .redx, .redy;
	}
	for ( .@i = .teamsize; .@i < .@n; ++.@i ) {
		attachrid .@ids[.@i];
		bg_join .blue, .arenaname$, .bluex, .bluey;
	}
	detachrid;

	.red_score = .blue_score = .teamsize;
	bg_updatescore .arenaname$, .red_score, .blue_score;
	mapannounce .arenaname$, "- As duas equipes entraram na Arena 4v4 (" + .arenaname$ + ")! -", bc_map;

	sleep .eventlasting * 1000;

	if ( .red_score > .blue_score ) {
		mapannounce .arenaname$, "- Time Vermelho venceu! -", bc_map;
		callsub L_Reward, .red;
	}
	else if ( .blue_score > .red_score ) {
		mapannounce .arenaname$, "- Time Azul venceu! -", bc_map;
		callsub L_Reward, .blue;
	}
	else
		mapannounce .arenaname$, "- A partida terminou empatada! -", bc_map;

	bg_warp .red, "izlude",125,164;
	bg_warp .blue, "izlude",129,164;
	bg_destroy .red;
	bg_destroy .blue;
	$@bg3v3_arena_busy[.@aidx] = 0;
	goto OnWaitOpen;

L_Reward:
	bg_get_data getarg(0), 1;
	for ( .@i = 0; .@i < $@arenamemberscount; ++.@i )
		getitem .rewarditem[0], .rewarditem[1], $@arenamembers[.@i];
	return;

OnRedDead:  callsub L_Dead, .red_score;
OnBlueDead: callsub L_Dead, .blue_score;
L_Dead:
	set getarg(0), getarg(0) -1;
	bg_updatescore .arenaname$, .red_score, .blue_score;
	bg_leave;
	if ( !getarg(0) )
		awake strnpcinfo(0);
	sleep2 1250;
	percentheal 100,100;
	end;

OnRedQuit:  callsub L_Quit, .red_score;
OnBlueQuit: callsub L_Quit, .blue_score;
L_Quit:
	set getarg(0), getarg(0) -1;
	bg_updatescore .arenaname$, .red_score, .blue_score;
	percentheal 100, 100;
	if ( !getarg(0) )
		awake strnpcinfo(0);
	end;
}
